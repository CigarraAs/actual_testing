            };

      } else if (type === 'string') {

            [aqlOp]: value,
          },

      case 'gte':
        return apply(field, '$gte', getValue(value));
      case 'lt':
        return apply(field, '$lt', getValue(value));
      case 'lte':
        return apply(field, '$lte', getValue(value));
      case 'true':

      case 'onBudget':
